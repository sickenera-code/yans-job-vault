// ---- DOM helper ----
function el(tag, props, ...children) {
  const node = document.createElement(tag);
  if (props) {
    for (const k in props) {
      const v = props[k];
      if (v == null) continue;
      if (k === "className") node.className = v;
      else if (k === "textContent") node.textContent = v;
      else if (k === "draggable") node.draggable = !!v;
      else if (k === "style") node.setAttribute("style", v);
      else if (k.startsWith("on") && typeof v === "function") {
        node.addEventListener(k.slice(2).toLowerCase(), v);
      } else {
        node.setAttribute(k, v);
      }
    }
  }
  for (const c of children) {
    if (c == null) continue;
    if (Array.isArray(c)) {
      c.forEach(x => {
        if (x == null) return;
        node.appendChild(typeof x === "string" ? document.createTextNode(x) : x);
      });
    } else if (typeof c === "string" || typeof c === "number") {
      node.appendChild(document.createTextNode(String(c)));
    } else {
      node.appendChild(c);
    }
  }
  return node;
}

// ---- 仓库内评级数据；不读取提供者或其他仓库的浏览器缓存 ----
const RANK_FILE = "Resource/岗位评级数据.md";
const RANK_PATTERN = /<!-- JOB_RANKS_DATA_START -->\s*```json\s*([\s\S]*?)\s*```\s*<!-- JOB_RANKS_DATA_END -->/;
let lastRankContent = null;
let rankBusy = false;
function emptyRankMap() {
  return Object.fromEntries(LEVELS.map(l => [l.key, []]));
}
function parseRankMap(content) {
  const match = content.match(RANK_PATTERN);
  if (!match) throw new Error("岗位评级数据标记缺失，已停止写入以保护原文件");
  const obj = JSON.parse(match[1]);
  if (!obj || Array.isArray(obj) || typeof obj !== "object") throw new Error("评级数据必须为对象");
  const out = emptyRankMap();
  for (const level of LEVELS) {
    if (!Array.isArray(obj[level.key]) || obj[level.key].some(id => typeof id !== "string")) {
      throw new Error("评级数据的每个等级必须为岗位路径数组");
    }
    out[level.key] = obj[level.key];
  }
  return out;
}
async function loadRankMap() {
  const file = app.vault.getAbstractFileByPath(RANK_FILE);
  if (!file) throw new Error("缺少 " + RANK_FILE + "，请重新运行技能初始化补齐文件");
  const content = await app.vault.read(file);
  const map = parseRankMap(content);
  lastRankContent = content;
  return map;
}
async function saveRankMap() {
  const file = app.vault.getAbstractFileByPath(RANK_FILE);
  if (!file) throw new Error("评级数据文件不存在");
  const out = emptyRankMap();
  LEVELS.forEach(l => { out[l.key] = state.ranks[l.key].map(it => it.id); });
  // process 原子检查旧内容，避免覆盖另一窗口或同步设备刚写入的评级。
  const updated = await app.vault.process(file, content => {
    if (content !== lastRankContent) throw new Error("评级数据已被其他窗口修改，请重新打开页面后重试");
    parseRankMap(content);
    const match = content.match(RANK_PATTERN);
    const data = {...JSON.parse(match[1]), ...out};
    const block = '<!-- JOB_RANKS_DATA_START -->\n```json\n' + JSON.stringify(data, null, 2) + '\n```\n<!-- JOB_RANKS_DATA_END -->';
    return content.replace(RANK_PATTERN, () => block);
  });
  lastRankContent = updated;
}
async function persistRanking(change) {
  if (rankBusy) return;
  rankBusy = true;
  const before = {pool: state.pool.slice(), ranks: Object.fromEntries(LEVELS.map(l => [l.key, state.ranks[l.key].slice()]))};
  try {
    change();
    await saveRankMap();
  } catch (error) {
    state.pool = before.pool;
    state.ranks = before.ranks;
    alert("评级未保存：" + error.message);
  } finally {
    rankBusy = false;
    renderFull();
  }
}

// ---- 数据操作 ----
function findHolder(itemId) {
  if (state.pool.some(x => x.id === itemId)) return "pool";
  for (const k of Object.keys(state.ranks)) {
    if (state.ranks[k].some(x => x.id === itemId)) return k;
  }
  return null;
}
async function moveItem(itemId, target) {
  if (target !== "pool" && !LEVELS.some(l => l.key === target)) return;
  const src = findHolder(itemId);
  if (!src || src === target) return;
  await persistRanking(() => {
    const source = src === "pool" ? state.pool : state.ranks[src];
    const item = source.find(x => x.id === itemId);
    if (src === "pool") state.pool = source.filter(x => x.id !== itemId);
    else state.ranks[src] = source.filter(x => x.id !== itemId);
    if (target === "pool") state.pool.push(item);
    else state.ranks[target].push(item);
  });
}
async function clearAllRanks() {
  const count = LEVELS.reduce((sum, l) => sum + state.ranks[l.key].length, 0);
  if (!count || !confirm("确认清除所有评级？\n将把 " + count + " 个已归类的方块移回候补池。")) return;
  await persistRanking(() => {
    LEVELS.forEach(l => {
      state.pool.push(...state.ranks[l.key]);
      state.ranks[l.key] = [];
    });
  });
}

// 从 dataview 同步拉取 已投递岗位/ 目录的页面
function loadJobsSync() {
  return dv.pages('"已投递岗位"')
    .where(p => p.company && p.seq)
    .sort(p => Number(p.seq), "asc")
    .array();
}
function pageToItem(p) {
  const company = p.company || "";
  let position = p.position || "";
  // 兜底：position 为空时，从 title 里剥掉公司名前缀推出岗位名
  if (!position && p.title) {
    position = String(p.title);
    if (company && position.indexOf(company) === 0) {
      position = position.slice(company.length).trim();
    }
  }
  return {
    id: p.file.path,                       // .md 路径作为稳定 id
    company: company,
    position: position,
    type: p.type || "",
    date: toDateStr(p.date),                // dataview 会把 ISO 日期解析为 Luxon DateTime 对象
    link: p.link || "",
    notePath: p.file.path
  };
}

// dataview 返回的日期字段可能是 Luxon DateTime 对象（带 toFormat / toISO 方法），
// 也可能是 null/undefined 或原始字符串，统一转成 "yyyy-MM-dd" 字符串
function toDateStr(v) {
  if (!v) return "";
  if (typeof v === "string") return v;
  if (typeof v === "number") return String(v);
  if (typeof v === "object" && v) {
    if (typeof v.toFormat === "function") {
      try { return v.toFormat("yyyy-MM-dd"); } catch (e) { /* fallthrough */ }
    }
    if (typeof v.toISODate === "function") {
      try { return v.toISODate(); } catch (e) { /* fallthrough */ }
    }
    if (typeof v.toISO === "function") {
      try { return String(v.toISO()).slice(0, 10); } catch (e) { /* fallthrough */ }
    }
  }
  return String(v);
}
function rebuildState(jobs, rankMap) {
  const itemsById = {};
  jobs.forEach(p => { const it = pageToItem(p); itemsById[it.id] = it; });

  const usedIds = new Set();
  const newRanks = {};
  LEVELS.forEach(l => { newRanks[l.key] = []; });
  for (const lk of Object.keys(rankMap || {})) {
    if (!newRanks[lk]) continue;
    for (const id of rankMap[lk]) {
      if (itemsById[id] && !usedIds.has(id)) {
        newRanks[lk].push(itemsById[id]);
        usedIds.add(id);
      }
    }
  }
  const newPool = [];
  for (const id of Object.keys(itemsById)) {
    if (!usedIds.has(id)) newPool.push(itemsById[id]);
  }
  newPool.sort((a, b) =>
    (b.date || "").localeCompare(a.date || "") ||
    (a.company || "").localeCompare(b.company || "", "zh")
  );
  // 上面那一行 .date 已经在 pageToItem 里 toDateStr 过，所以都是字符串了，
  // 这里安全调用 .localeCompare
  state.pool = newPool;
  state.ranks = newRanks;
}

// ---- 渲染组件 ----
function safeLen(arr) { return (arr && arr.length) || 0; }

function makeCard(it) {
  const card = el("div", { className: "thl-card", draggable: true, title: (it.date || "") + "  " + (it.notePath || "") },
    el("div", { className: "co", textContent: it.company || "(无名称)" }),
    el("div", { className: "po", textContent: it.position || "—" })
  );
  const metaParts = [];
  if (it.date) metaParts.push("📅 " + it.date);
  if (it.link) metaParts.push("🔗");
  if (metaParts.length) card.appendChild(el("div", { className: "meta", textContent: metaParts.join(" · ") }));
  card.addEventListener("dragstart", e => {
    e.dataTransfer.setData("text/plain", it.id);
    e.dataTransfer.effectAllowed = "move";
  });
  card.addEventListener("dblclick", () => {
    if (it.notePath) {
      const f = app.vault.getAbstractFileByPath(it.notePath);
      if (f) app.workspace.openLinkText(it.notePath, "", false);
    }
  });
  return card;
}
function makeColumn(level) {
  const col = el("div", { className: "thl-col", "data-key": level.key },
    el("div", { className: "thl-col-head" },
      el("span", { textContent: level.emoji }),
      el("span", { textContent: level.key }),
      el("span", { style: "margin-left:auto;font-size:12px;opacity:0.85", textContent: String(safeLen(state.ranks[level.key])) })
    )
  );
  const arr = state.ranks[level.key] || [];
  if (arr.length === 0) {
    col.appendChild(el("div", { className: "thl-empty", textContent: "空 · 拖动到此处" }));
  } else {
    arr.forEach(it => col.appendChild(makeCard(it)));
  }
  col.addEventListener("dragover", e => { e.preventDefault(); col.classList.add("drag-over"); });
  col.addEventListener("dragleave", () => col.classList.remove("drag-over"));
  col.addEventListener("drop", e => {
    e.preventDefault();
    col.classList.remove("drag-over");
    const id = e.dataTransfer.getData("text/plain");
    if (id) moveItem(id, level.key);
  });
  return col;
}
function makeBoard() {
  return el("div", { className: "thl-board" }, ...LEVELS.map(makeColumn));
}

}

// 确保容器里存在 #thl-root（不存在则创建），再整体重绘
function ensureRoot() {
  let root = dv.container.querySelector("#thl-root");
  if (!root) {
    dv.container.innerHTML = "";
    root = el("div", { className: "thl", id: "thl-root" });
    dv.container.appendChild(root);
  }
  return root;
}
function renderFull() {
  const root = ensureRoot();
  root.innerHTML = "";
  root.appendChild(makeBanner());
  root.appendChild(makeBoard());
  root.appendChild(makePool());
}
function renderError(msg, stack) {
  dv.container.innerHTML = "";
  const wrap = el("div", { className: "thl-err" });
  wrap.appendChild(el("div", { style: "font-weight:600;margin-bottom:6px", textContent: "⚠️ 加载失败：" + msg }));
  if (stack) {
    const pre = el("pre", { style: "white-space:pre-wrap;font-size:11px;opacity:0.75;background:rgba(0,0,0,.04);padding:8px;border-radius:6px;max-height:280px;overflow:auto" });
    pre.textContent = stack;
    wrap.appendChild(pre);
  }
  wrap.appendChild(el("div", { style: "margin-top:8px;opacity:.7;font-size:11px", textContent: "💡 提示：按 ⌘/Ctrl + Shift + I 打开 DevTools → Console 可查看更多上下文" }));
  dv.container.appendChild(wrap);
}

// ---- 加载岗位索引及仓库内评级，然后渲染 ----
async function loadAndRender() {
  try {
    const jobs = loadJobsSync();
    const rankMap = await loadRankMap();
    rebuildState(jobs, rankMap);
    renderFull();
  } catch (err) {
    console.error("[从夯到拉] loadAndRender", err);
    renderError(err.message || String(err), err.stack);
  }
}

// ---- 启动：岗位笔记变化由 Dataview 索引，重新渲染时从仓库读取评级 ----
ensureRoot();
loadAndRender();
