// 数据加载与保存（仓库内 Markdown 文件为唯一数据源）
// ============================================
const DATA_FILE = 'Resource/秋招关注数据.md';
const DATA_START_MARKER = '<!-- WATCH_POSITIONS_DATA_START -->';
const DATA_END_MARKER = '<!-- WATCH_POSITIONS_DATA_END -->';
const DATA_BLOCK_PATTERN = /<!-- WATCH_POSITIONS_DATA_START -->\s*```json\s*([\s\S]*?)\s*```\s*<!-- WATCH_POSITIONS_DATA_END -->/;
let saveQueue = Promise.resolve();

function buildDataBlock(serializedData) {
    return `${DATA_START_MARKER}\n\`\`\`json\n${serializedData}\n\`\`\`\n${DATA_END_MARKER}`;
}

function buildDataFile(serializedData) {
    return `---
title: 秋招关注数据
tags:
  - 求职
  - 岗位关注
---

# 秋招关注数据

> [!warning] 自动维护文件
> 此文件由 [[秋招关注]] 自动读写。请勿修改下面的数据标记或 JSON 结构；请将本文件与仓库一起同步或备份。

${buildDataBlock(serializedData)}
`;
}

function parseDataFile(content) {
    const match = content.match(DATA_BLOCK_PATTERN);
    if (!match) {
        throw new Error(`未在 ${DATA_FILE} 中找到有效的数据区块`);
    }
    const parsed = JSON.parse(match[1]);
    if (!Array.isArray(parsed)) {
        throw new Error(`${DATA_FILE} 中的数据必须是数组`);
    }
    return parsed;
}

async function writeDataSnapshot(serializedData) {
    const dataFile = app.vault.getAbstractFileByPath(DATA_FILE);
    if (!dataFile) {
        await app.vault.create(DATA_FILE, buildDataFile(serializedData));
        return;
    }
    const content = await app.vault.read(dataFile);
    if (!DATA_BLOCK_PATTERN.test(content)) {
        throw new Error(`未在 ${DATA_FILE} 中找到有效的数据区块，已停止写入以保护原文件`);
    }
    const updatedContent = content.replace(DATA_BLOCK_PATTERN, () => buildDataBlock(serializedData));
    await app.vault.modify(dataFile, updatedContent);
}

function saveData() {
    const serializedData = JSON.stringify(watchPositions, null, 2);
    const saveOperation = saveQueue
        .catch(() => {})
        .then(() => writeDataSnapshot(serializedData));
    saveQueue = saveOperation;
    return saveOperation;
}

async function loadData() {
    const dataFile = app.vault.getAbstractFileByPath(DATA_FILE);
    watchPositions = dataFile
        ? parseDataFile(await app.vault.read(dataFile))
        : [];

    // 兼容旧数据：原“已过期”记录归入“关注中”
    let needsSave = false;
    watchPositions.forEach(pos => {
        if (!pos.id) { pos.id = Date.now().toString() + Math.random().toString(36).substr(2, 5); needsSave = true; }
        if (!pos.status || pos.status === 'expired') { pos.status = 'watching'; needsSave = true; }
        if (!pos.stages) { pos.stages = []; }
    });
    if (needsSave) await saveData();
}

// 工具函数
// ============================================
function getMonthsUntil(monthStr) {
    if (!monthStr) return null;
    const [year, month] = monthStr.split('-').map(Number);
    const today = new Date();
    return (year - today.getFullYear()) * 12 + (month - 1 - today.getMonth());
}

function getCountdownInfo(monthStr) {
    const months = getMonthsUntil(monthStr);
    if (months === null) return { text: '未设置', cls: 'past' };
    if (months > 2) return { text: `${months}个月后`, cls: 'far' };
    if (months > 1) return { text: `${months}个月后`, cls: 'soon' };
    if (months === 1) return { text: '下月开放', cls: 'imminent' };
    if (months === 0) return { text: '本月开放', cls: 'today' };
    return { text: `已过${Math.abs(months)}个月`, cls: 'past' };
}

function formatDateStamp(monthStr) {
    if (!monthStr) return { year: '—', month: '—' };
    const [year, month] = monthStr.split('-');
    return {
        year,
        month: `${Number(month)}月`
    };
}

function escapeHtml(text) {
    if (!text) return '';
    const div = document.createElement('div');
    div.textContent = text;
    return div.innerHTML;
}

function getStats() {
    const total = watchPositions.length;
    const watching = watchPositions.filter(p => p.status === 'watching').length;
    const opened = watchPositions.filter(p => p.status === 'opened').length;

    // 即将开放：7天内
    const upcoming = watchPositions.filter(p => {
        if (p.status !== 'watching') return false;
        const months = getMonthsUntil(p.expectedDate);
        return months !== null && months >= 0 && months <= 1;
    }).length;

    return { total, watching, opened, upcoming };
}

        `;
    }

    // 需要获取过滤后元素在原数组中的真实索引

    }).join('');
}

window.watchAddPosition = async function(event) {
    event.preventDefault();

    const company = document.getElementById('watchCompanyInput').value.trim();
    const type = document.getElementById('watchTypeInput').value;
    const expectedDate = document.getElementById('watchDateInput').value;
    const url = document.getElementById('watchUrlInput').value.trim();
    const note = document.getElementById('watchNoteInput').value.trim();

    if (!company || !expectedDate) {
        alert('请填写必填字段');
        return;
    }

    const newPos = {
        id: Date.now().toString(),
        company,
        type,
        expectedDate,
        url: url || '',
        status: 'watching',
        note: note || ''
    };

    watchPositions.push(newPos);
    try {
        await saveData();
        render();
    } catch (error) {
        watchPositions.pop();
        console.error('关注岗位保存失败：', error);
        alert(`保存失败：${error.message}`);
    }
};

window.watchDeletePosition = async function(id) {
    const index = watchPositions.findIndex(p => p.id === id);
    if (index === -1) return;

    const pos = watchPositions[index];
    if (!confirm(`确定要删除「${pos.company}」吗？此操作不可恢复。`)) return;

    const removed = watchPositions.splice(index, 1)[0];
    try {
        await saveData();
        render();
    } catch (error) {
        watchPositions.splice(index, 0, removed);
        console.error('删除失败：', error);
        alert(`删除失败：${error.message}`);
    }
};

window.watchHandleDragStart = function(e, index) {
    draggedIndex = index;
    e.currentTarget.classList.add('dragging');
    e.dataTransfer.effectAllowed = 'move';
    e.dataTransfer.setData('text/plain', String(index));
};

window.watchHandleDragOver = function(e, index) {
    e.preventDefault();
    e.dataTransfer.dropEffect = 'move';
    if (draggedIndex === null || draggedIndex === index) return;

    const card = e.currentTarget;
    // 清除其他卡片的拖拽悬停样式
    document.querySelectorAll('.watch-card.drag-over').forEach(el => {
        if (el !== card) el.classList.remove('drag-over', 'drag-over-top', 'drag-over-bottom');
    });

    card.classList.add('drag-over');
    // 判断鼠标在卡片上半还是下半
    const rect = card.getBoundingClientRect();
    const midY = rect.top + rect.height / 2;
    if (e.clientY < midY) {
        card.classList.add('drag-over-top');
        card.classList.remove('drag-over-bottom');
    } else {
        card.classList.add('drag-over-bottom');
        card.classList.remove('drag-over-top');
    }
};

window.watchHandleDragLeave = function(e) {
    // 只有真正离开卡片才移除样式
    if (!e.currentTarget.contains(e.relatedTarget)) {
        e.currentTarget.classList.remove('drag-over', 'drag-over-top', 'drag-over-bottom');
    }
};

window.watchHandleDrop = async function(e, index) {
    e.preventDefault();
    e.stopPropagation();

    if (draggedIndex === null || draggedIndex === index) {
        draggedIndex = null;
        return;
    }

    // 根据鼠标位置决定插入点
    const card = e.currentTarget;
    const rect = card.getBoundingClientRect();
    const midY = rect.top + rect.height / 2;
    let insertIndex = e.clientY < midY ? index : index + 1;

    // 如果从上往下拖，需要调整索引
    if (draggedIndex < insertIndex) {
        insertIndex--;
    }

    const item = watchPositions.splice(draggedIndex, 1)[0];
    watchPositions.splice(insertIndex, 0, item);

    draggedIndex = null;
    try {
        await saveData();
        render();
    } catch (error) {
        console.error('拖拽排序失败：', error);
        alert(`排序失败：${error.message}`);
        render();
    }
};

window.watchHandleDragEnd = function(e) {
    e.currentTarget.classList.remove('dragging');
    document.querySelectorAll('.watch-card.drag-over, .watch-card.drag-over-top, .watch-card.drag-over-bottom').forEach(el => {
        el.classList.remove('drag-over', 'drag-over-top', 'drag-over-bottom');
    });
    draggedIndex = null;
};

window.watchConfirmImport = function() {
    const fileInput = document.getElementById('watchImportFileInput');
    const file = fileInput.files[0];
    if (!file) {
        alert('请选择要导入的文件');
        return;
    }

    const reader = new FileReader();
    reader.onload = async function(e) {
        try {
            const importedData = JSON.parse(e.target.result);
            if (!Array.isArray(importedData)) {
                alert('文件格式错误：数据不是有效的数组格式');
                return;
            }
            if (confirm(`确定要导入 ${importedData.length} 条记录吗？这将覆盖当前所有数据。`)) {
                const prev = watchPositions;
                watchPositions = importedData;
                try {
                    await saveData();
                    render();
                    watchCloseModal();
                    alert('数据导入成功！');
                } catch (error) {
                    watchPositions = prev;
                    console.error('导入失败：', error);
                    alert(`导入失败：${error.message}`);
                }
            }
        } catch (error) {
            alert('文件解析失败：' + error.message);
        }
    };
    reader.readAsText(file);
};
