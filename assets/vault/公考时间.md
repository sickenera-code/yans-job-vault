// 渲染函数
// ============================================
function renderTimeline() {
    const container = dv.container;
    container.innerHTML = '';
    container.className = 'exam-container';

    // 计算总任务数
    let totalTasks = 0;
    timelineData.forEach(yearData => {
        yearData.months.forEach(monthData => {
            totalTasks += monthData.tasks.length;
        });
    });

    // --- 樱花飘落 ---
    const sakuraContainer = document.createElement('div');
    sakuraContainer.className = 'sakura-container';
    for (let i = 0; i < 10; i++) {
        sakuraContainer.appendChild(document.createElement('div')).className = 'sakura';
    }
    container.appendChild(sakuraContainer);

    // --- 枫叶飘落 ---
    const momijiContainer = document.createElement('div');
    momijiContainer.className = 'momiji-container';
    for (let i = 0; i < 5; i++) {
        momijiContainer.appendChild(document.createElement('div')).className = 'momiji';
    }
    container.appendChild(momijiContainer);

    // --- 头部 ---
    const header = document.createElement('div');
    header.className = 'exam-header';
    header.innerHTML = `
        <h1 class="exam-title">公考巡考时间轴</h1>
        <p class="exam-subtitle">2026.1 — 2027.6 · 2027届应届生版</p>
        <div class="exam-divider"></div>
    `;
    container.appendChild(header);

    // --- 总结卡片 ---
    const summary = document.createElement('div');
    summary.className = 'exam-summary';
    summary.innerHTML = `
        <div class="summary-card">
            <span class="summary-icon">🍂</span>
            <div class="summary-period">2026.9 — 12</div>
            <div class="summary-content">国考、定向选调<br>下半年省考、军队文职</div>
        </div>
        <div class="summary-card">
            <span class="summary-icon">🌸</span>
            <div class="summary-period">2027.2 — 3</div>
            <div class="summary-content">2027上半年省考联考<br>最重要的考试</div>
        </div>
        <div class="summary-card">
            <span class="summary-icon">🌿</span>
            <div class="summary-period">2027.4 — 6</div>
            <div class="summary-content">三支一扶、事业单位<br>西部计划、特岗教师</div>
        </div>
    `;
    container.appendChild(summary);

    // --- 进度条 ---
    const progressSection = document.createElement('div');
    progressSection.className = 'progress-section';
    progressSection.innerHTML = `
        <div class="progress-bar">
            <div class="progress-title">备考进度</div>
            <div class="progress-track">
                <div class="progress-fill" style="width: 0%"></div>
            </div>
            <div class="progress-text">
                已完成 <span class="progress-number">0</span> / <span class="progress-total">${totalTasks}</span> 项任务
            </div>
        </div>
    `;
    container.appendChild(progressSection);

    // --- 时间轴 ---
    const timelineWrapper = document.createElement('div');
    timelineWrapper.className = 'timeline-wrapper';

    // 连接线
    timelineWrapper.appendChild(Object.assign(document.createElement('div'), { className: 'timeline-connector' }));

    // 渲染每年的数据
    let nodeIndex = 0;
    timelineData.forEach((yearData, yearIdx) => {
        const yearSection = document.createElement('div');
        yearSection.className = 'year-section';

        // 年份标题
        const yearTitle = document.createElement('h2');
        yearTitle.className = 'year-title';
        yearTitle.textContent = yearData.year + '年';
        yearTitle.style.animationDelay = `${0.5 + yearIdx * 0.2}s`;
        yearSection.appendChild(yearTitle);

        // 月份节点容器
        const pathContainer = document.createElement('div');
        pathContainer.className = 'timeline-path';

        yearData.months.forEach((monthData, monthIdx) => {
            const examType = EXAM_TYPES[monthData.type] || EXAM_TYPES.beikao;

            const node = document.createElement('div');
            node.className = 'timeline-node';
            node.style.animationDelay = `${0.6 + nodeIndex * 0.12}s`;

            const tasksHtml = monthData.tasks.map((task, taskIdx) => `
                <li class="node-task" data-task="${yearIdx}-${monthIdx}-${taskIdx}">
                    <span class="task-checkbox" role="checkbox" tabindex="0"></span>
                    <span class="task-text">${task}</span>
                </li>
            `).join('');

            const cardClasses = [
                'node-card',
                monthData.highlight ? 'highlight' : '',
                monthData.status === 'current' ? 'current' : ''
            ].filter(Boolean).join(' ');

            node.innerHTML = `
                <div class="node-dot" style="--dot-color: ${examType.color};"></div>
                <div class="${cardClasses}" style="--exam-color: ${examType.color};">
                    ${monthData.important ? '<div class="important-badge"></div>' : ''}
                    <span class="node-month">${monthData.month}</span>
                    <h3 class="node-title">
                        <span class="node-icon">${examType.icon}</span>
                        ${monthData.title}
                    </h3>
                    <ul class="node-tasks">
                        ${tasksHtml}
                    </ul>
                    <div class="node-note">${monthData.note}</div>
                </div>
            `;

            pathContainer.appendChild(node);
            nodeIndex++;
        });

        yearSection.appendChild(pathContainer);
        timelineWrapper.appendChild(yearSection);
    });

    container.appendChild(timelineWrapper);

    // --- 打卡交互（事件委托） ---
    function updateProgress() {
        const allCheckboxes = container.querySelectorAll('.task-checkbox');
        const checkedCount = container.querySelectorAll('.task-checkbox.checked').length;
        const percent = allCheckboxes.length > 0
            ? (checkedCount / allCheckboxes.length * 100).toFixed(0)
            : 0;
        const fill = container.querySelector('.progress-fill');
        const num = container.querySelector('.progress-number');
        if (fill) fill.style.width = percent + '%';
        if (num) num.textContent = checkedCount;
    }

    container.querySelectorAll('.task-checkbox').forEach(checkbox => {
        // 点击切换
        checkbox.addEventListener('click', function(e) {
            e.stopPropagation();
            this.classList.toggle('checked');
            this.closest('.node-task').classList.toggle('checked');
            this.setAttribute('aria-checked', this.classList.contains('checked'));
            updateProgress();
        });
        // 键盘可访问
        checkbox.addEventListener('keydown', function(e) {
            if (e.key === ' ' || e.key === 'Enter') {
                e.preventDefault();
                this.click();
            }
        });
    });
}

// 执行渲染
renderTimeline();
