// 数据加载与保存（仓库内 Markdown 文件为唯一数据源）

const DATA_FILE = 'Resource/投递记录数据.md';
const DATA_START_MARKER = '<!-- JOB_APPLICATIONS_DATA_START -->';
const DATA_END_MARKER = '<!-- JOB_APPLICATIONS_DATA_END -->';
const DATA_BLOCK_PATTERN = /<!-- JOB_APPLICATIONS_DATA_START -->\s*```json\s*([\s\S]*?)\s*```\s*<!-- JOB_APPLICATIONS_DATA_END -->/;
let saveQueue = Promise.resolve();

function buildDataBlock(serializedData) {
    return `${DATA_START_MARKER}\n\`\`\`json\n${serializedData}\n\`\`\`\n${DATA_END_MARKER}`;
}

function buildDataFile(serializedData) {
    return `---
title: 投递记录数据
tags:
  - 求职
  - 投递记录
---

# 投递记录数据

> [!warning] 自动维护文件
> 此文件由 [[投递进度]] 自动读写。请勿修改下面的数据标记或 JSON 结构；请将本文件与仓库一起同步或备份。

${buildDataBlock(serializedData)}
`;
}

function validateApplications(records) {
    const ids = new Set();
    for (const record of records) {
        if (!record || typeof record !== 'object') throw new Error('投递条目必须是对象');
        for (const key of ['id', 'company', 'position', 'date', 'type']) {
            if (typeof record[key] !== 'string' || !record[key].trim()) throw new Error(`投递记录缺少 ${key}`);
        }
        if (!/^[A-Za-z0-9_-]+$/.test(record.id) || ids.has(record.id)) throw new Error('投递 id 无效或重复');
        ids.add(record.id);
        if (!COMPANY_TYPES[record.type] && record.type !== 'waiqi') throw new Error('企业类型无效');
        if (!/^\d{4}-\d{2}-\d{2}$/.test(record.date) || Number.isNaN(Date.parse(record.date)) || new Date(record.date).toISOString().slice(0, 10) !== record.date) throw new Error('投递日期无效');
        if (record.stages !== undefined && (!Array.isArray(record.stages) || record.stages.some(s => typeof s !== 'string'))) throw new Error('stages 必须是字符串数组');
        for (const key of ['offer', 'terminated']) {
            if (record[key] !== undefined && typeof record[key] !== 'boolean') throw new Error(`${key} 必须是布尔值`);
        }
        if (record.link && (typeof record.link !== 'string' || !/^https?:\/\/\S+$/i.test(record.link))) throw new Error('岗位链接必须为 HTTP(S) 地址');
        if (record.notePath && (typeof record.notePath !== 'string' || !/^已投递岗位\/[^/\\]+\.md$/.test(record.notePath))) throw new Error('岗位笔记路径无效');
    }
    return records;
}

function parseDataFile(content) {
    const match = content.match(DATA_BLOCK_PATTERN);
    if (!match) {
        throw new Error(`未在 ${DATA_FILE} 中找到有效的数据区块`);
    }

    const parsed = JSON.parse(match[1]);
    if (!Array.isArray(parsed)) {
        throw new Error(`${DATA_FILE} 中的数据必须是数组`);
    }
    return validateApplications(parsed);
}

async function writeDataSnapshot(serializedData) {
    const dataFile = app.vault.getAbstractFileByPath(DATA_FILE);
    if (!dataFile) {
        await app.vault.create(DATA_FILE, buildDataFile(serializedData));
        return;
    }

    const content = await app.vault.read(dataFile);
    if (!DATA_BLOCK_PATTERN.test(content)) {
        throw new Error(`未在 ${DATA_FILE} 中找到有效的数据区块，已停止写入以保护原文件`);
    }

    const updatedContent = content.replace(DATA_BLOCK_PATTERN, () => buildDataBlock(serializedData));
    await app.vault.modify(dataFile, updatedContent);
}

function saveData() {
    const serializedData = JSON.stringify(applications, null, 2);
    const saveOperation = saveQueue
        .catch(() => {})
        .then(() => writeDataSnapshot(serializedData));
    saveQueue = saveOperation;
    return saveOperation;
}

async function loadData() {
    const dataFile = app.vault.getAbstractFileByPath(DATA_FILE);
    let shouldSave = !dataFile;

    applications = dataFile
        ? parseDataFile(await app.vault.read(dataFile))
        : [];

    // 兼容旧版重复的“外企”类型，并统一迁移为 foreign。
    applications.forEach(application => {
        if (!application.stages) { application.stages = []; shouldSave = true; }
        if (application.type === 'waiqi') {
            application.type = 'foreign';
            shouldSave = true;
        }
    });

    if (shouldSave) {
        await saveData();
    }


}

}

let noteQueue = Promise.resolve();
function syncJobNote(record) {
    const task = noteQueue.catch(() => {}).then(() => ensureJobWorkspace(record));
    noteQueue = task;
    return task;
}

async function ensureJobWorkspace(record) {
    const vault = app.vault;
    let folder = vault.getAbstractFileByPath(NOTE_FOLDER);
    if (!folder) {
        await vault.createFolder(NOTE_FOLDER);
        folder = vault.getAbstractFileByPath(NOTE_FOLDER);
    }
    if (!folder || !Array.isArray(folder.children)) throw new Error('已投递岗位路径不是文件夹');
    let filePath = record.notePath;
    if (filePath && !/^已投递岗位\/[^/\\]+\.md$/.test(filePath)) {
        throw new Error('notePath 必须指向已投递岗位下的直接子 Markdown 文件');
    }
    // 用 application_id 恢复已创建但尚未回写数据文件的笔记，避免重试产生重复文件。
    if (!filePath) {
        for (const child of folder.children) {
            if (!child.name.endsWith('.md')) continue;
            const content = await vault.read(child);
            const idLine = `application_id: ${JSON.stringify(String(record.id))}`;
            if (content.split('\n').includes(idLine)) { filePath = child.path; break; }
        }
    }
    const used = folder.children.map(c => Number(c.name.match(/^(\d+)-/)?.[1] || 0));
    const seq = filePath?.split('/').pop().match(/^(\d+)-/)?.[1]
        || String(Math.max(0, ...used) + 1).padStart(2, '0');
    const note = buildJobNoteContent(record, seq, filePath);
    const existing = vault.getAbstractFileByPath(note.filePath);
    if (existing && Array.isArray(existing.children)) throw new Error('岗位笔记路径被文件夹占用');
    if (existing) {
        const content = await vault.read(existing);
        const owner = content.match(/^application_id: (.+)$/m);
        if (owner && JSON.parse(owner[1]) !== String(record.id)) throw new Error('该笔记已属于其他投递 id');
    }
    if (!existing) await vault.create(note.filePath, note.body);
    let materialFolder = vault.getAbstractFileByPath(note.folderPath);
    if (!materialFolder) materialFolder = await vault.createFolder(note.folderPath);
    if (!Array.isArray(materialFolder.children)) throw new Error('岗位资料路径被文件占用');
    const indexPath = `${note.folderPath}/00-资料索引.md`;
    if (!vault.getAbstractFileByPath(indexPath)) await vault.create(indexPath, note.indexBody);
    record.notePath = note.filePath;
    record.folderPath = note.folderPath;
    return { filePath: note.filePath, folderPath: note.folderPath };
}

async function syncAllJobNotes() {
    let changed = false;
    const failures = [];
    for (const record of applications) {
        const before = JSON.stringify([record.notePath, record.folderPath]);
        try {
            await syncJobNote(record);

}

async function updateJobNoteOffer(record) {
    if (!record.notePath) return;
    const note = app.vault.getAbstractFileByPath(record.notePath);
    if (!note) return;
    const content = await app.vault.read(note);
    const updated = content.includes('offer:')
        ? content.replace(/^offer:.*$/m, 'offer: "true"')
        : content.replace(/^(type:.*)$/m, '$1\noffer: "true"');
    const body = updated.includes('## 投递进度')
        ? updated.replace('## 投递进度', '## 投递进度\n\n> [!success] 已获得 Offer\n> 已进入 Offer 归档')
        : updated;
    await app.vault.modify(note, body);
}

async function updateJobNoteLink(record) {
    if (!record.notePath) return;
    const note = app.vault.getAbstractFileByPath(record.notePath);
    if (!note) return;
    const content = await app.vault.read(note);
    const yamlEscape = value => JSON.stringify(String(value ?? '')).slice(1, -1);
    const linkLine = `link: "${yamlEscape(record.link || '')}"`;
    const linkBlockPattern = /^link:\s*".*"$/m;

    let updated;
    if (linkBlockPattern.test(content)) {
        updated = content.replace(linkBlockPattern, linkLine);
    } else {
        // 插入到 offer 之后；如果 offer 也不存在，则插入到 seq 之前
        if (/^offer:\s*".*"$/m.test(content)) {
            updated = content.replace(/^(offer:\s*".*")$/m, `$1\n${linkLine}`);
        } else {
            updated = content.replace(/^(seq:\s*".*")$/m, `$1\n${linkLine}`);
        }
    }

    // 同步正文中的「岗位链接」callout 块
    const linkCalloutPattern = /\n?> \[!link\] 岗位链接\n> \[点击跳转\]\([^\)]*\)\n?/;
    if (record.link) {
        const newBlock = `\n> [!link] 岗位链接\n> [点击跳转](${record.link})\n`;
        updated = linkCalloutPattern.test(updated)
            ? updated.replace(linkCalloutPattern, newBlock)
            : updated.replace(/(> \[!info\][\s\S]*?(?=\n## ))/, (match) => `${match}${newBlock}`);
    } else if (linkCalloutPattern.test(updated)) {
        updated = updated.replace(linkCalloutPattern, '\n');
    }

    await app.vault.modify(note, updated);
}

// HTML 转义，避免用户输入触发 XSS 或破坏卡片结构
function escapeHtml(value) {
    return String(value ?? '').replace(/[&<>"']/g, ch => ({
        '&': '&amp;',
        '<': '&lt;',
        '>': '&gt;',
        '"': '&quot;',
        "'": '&#39;'
    })[ch]);
}

// 链接规范化：自动补全协议；空值返回空字符串
function normalizeLink(link) {
    if (!link) return '';
    const trimmed = String(link).trim();
    if (!trimmed) return '';
    if (/^https?:\/\//i.test(trimmed)) return trimmed;
    return `https://${trimmed}`;
}

function renderProgress(app) {
    let stages = app.stages || [];
    const isInitialPlaceholder = !app.terminated && !app.offer && stages.length === 0;
    const displayStages = isInitialPlaceholder
        ? ['投递']
        : (app.offer ? stages : (app.terminated ? [...stages, '终止'] : stages));

    const currentStage = app.offer
        ? 'offer'
        : (app.terminated
            ? '已终止'
            : (isInitialPlaceholder ? '等待添加' : displayStages[displayStages.length - 1]));
    const progressLabel = app.offer
        ? `已获得 offer，共 ${stages.length} 个阶段`
        : (app.terminated
            ? `投递流程已终止，共 ${stages.length} 个阶段`
            : `投递流程，当前阶段：${currentStage}`);

    let progressHTML = `
        <nav class="job-progress" aria-label="${progressLabel}">
            <ol class="job-progress-list">
    `;

    displayStages.forEach((stage, index) => {
        let dotClass = '';
        let containerClass = '';
        let label = stage;
        const isLast = index === displayStages.length - 1;
        const isTerminalNode = (app.terminated || app.offer) && isLast;

        if (isTerminalNode) {
            dotClass = app.offer ? 'offer' : 'terminated';
            containerClass = dotClass;
        } else if (isInitialPlaceholder) {
            dotClass = 'initial';
            containerClass = 'initial';
        } else if ((app.terminated || app.offer) && index < stages.length) {
            dotClass = 'active';
            containerClass = 'active';
        } else if (index < displayStages.length - 1) {
            dotClass = 'active';
            containerClass = 'active';
        } else {
            dotClass = 'current';
            containerClass = 'current';
        }

        const hintBadge = isInitialPlaceholder
            ? `<span class="job-progress-empty-hint">+</span>`
            : '';
        const stateText = isTerminalNode
            ? (app.offer ? '已获得 offer' : '已终止')
            : (isInitialPlaceholder ? '待添加' : (dotClass === 'current' ? '当前阶段' : '已完成'));

        progressHTML += `
            <li class="job-progress-dot-container ${containerClass}" aria-label="${label}，${stateText}" ${dotClass === 'current' ? 'aria-current="step"' : ''}>
                ${hintBadge}
                <span class="job-progress-dot ${dotClass}" aria-hidden="true"></span>
                <span class="job-progress-label">${label}</span>
            </li>
        `;

        if (!isLast) {
            const connectorClass = (app.terminated || app.offer) && index === displayStages.length - 2
                ? (app.offer ? 'offer' : 'terminated')
                : 'active';
            progressHTML += `<li class="job-progress-connector ${connectorClass}" aria-hidden="true"></li>`;
        }
    });

    progressHTML += `
            </ol>
            <div class="job-progress-meta" aria-hidden="true">
                <span>${app.offer || app.terminated ? '流程状态' : '当前阶段'}</span>
                <strong>${currentStage}</strong>
            </div>
        </nav>
    `;
    return progressHTML;
}

// ============================================
function renderContent() {
    const typeFilter = dv.container.querySelector('#typeFilter')?.value || 'all';
    const statusFilter = dv.container.querySelector('.job-filter-btn.active')?.dataset.status || 'active';
    
    let filtered = [...applications];
    
    if (typeFilter !== 'all') {
        filtered = filtered.filter(app => app.type === typeFilter);
    }
    
    if (statusFilter === 'active') {
        filtered = filtered.filter(app => !app.terminated && !app.offer);
    } else if (statusFilter === 'offer') {
        filtered = filtered.filter(app => app.offer);
    } else if (statusFilter === 'terminated') {
        filtered = filtered.filter(app => app.terminated && !app.offer);
    }
    
    // 排序：进行中的在前，Offer 与已终止的归档状态在后
    filtered.sort((a, b) => {
        const statusOrder = app => app.offer ? 2 : (app.terminated ? 3 : 1);
        if (statusOrder(a) !== statusOrder(b)) {
            return statusOrder(a) - statusOrder(b);
        }
        return new Date(b.date) - new Date(a.date);
    });
    
    if (filtered.length === 0) {
        return `
            <div class="job-empty">
                <div class="job-empty-icon">📭</div>
                <div class="job-empty-text">暂无投递记录</div>
            </div>
        `;
    }
    
    return filtered.map((app, index) => renderCard(app, index)).join('');
}

window.addApplication = async function(event) {
    event.preventDefault();

    const form = event.currentTarget;
    const company = form.querySelector('#companyInput')?.value.trim() || '';
    const type = form.querySelector('#typeInput')?.value || '';
    const position = form.querySelector('#positionInput')?.value.trim() || '';
    const date = form.querySelector('#dateInput')?.value.trim() || '';
    const link = form.querySelector('#linkInput')?.value.trim() || '';

    if (!company || !position || !type || !date) {
        alert('请完整填写企业名称、企业类型、岗位名称和投递日期');
        return;
    }
    
    const newApp = {
        id: crypto.randomUUID(),
        company,
        type,
        position,
        date,
        link: normalizeLink(link),
        stages: [],
        terminated: false,
        offer: false
    };
    
    applications.push(newApp);
    try {
        await saveData();
    } catch (error) {
        applications = applications.filter(item => item.id !== newApp.id);
        alert(`保存失败：${error.message}`);
        return;
    }
    // 记录持久化后独立建目录。失败保留已保存记录，重开页面会重试。
    try {
        await syncJobNote(newApp);
        await saveData();
    } catch (error) {
        alert(`已保存投递记录，岗位资料同步未完成：${error.message}。重新打开页面可重试。`);
    }
    render();
};

window.markOffer = async function(id) {
    const app = applications.find(a => a.id === id);
    if (!app || app.offer) return;
    if (!confirm(`确认 ${app.company} 已获得 Offer 吗？`)) return;

    const previousOffer = app.offer;
    const previousTerminated = app.terminated;
    app.offer = true;
    app.terminated = false;
    if (!app.stages) app.stages = [];
    if (app.stages[app.stages.length - 1] !== 'offer') {
        app.stages.push('offer');
    }
    try {
        await saveData();
        await updateJobNoteOffer(app);
        render();
    } catch (error) {
        app.offer = previousOffer;
        app.terminated = previousTerminated;
        console.error('Offer 状态保存失败：', error);
        alert(`保存失败：${error.message}`);
    }
};

window.terminateProgress = async function(id) {

            render();

            alert(`保存失败：${error.message}`);
        }
    }
};

                alert(`删除失败：${error.message}`);
            }
        }
    }
};

window.confirmImport = function() {
    const fileInput = document.getElementById('importFileInput');
    const file = fileInput.files[0];
    
    if (!file) {
        alert('请选择要导入的文件');
        return;
    }
    
    const reader = new FileReader();
    reader.onload = async function(e) {
        try {
            const importedData = JSON.parse(e.target.result);
            if (Array.isArray(importedData)) {
                validateApplications(importedData);
                importedData.forEach(record => {
                    if (record.type === 'waiqi') record.type = 'foreign';
                    if (!record.stages) record.stages = [];
                });
            }
            if (Array.isArray(importedData)) {
                if (confirm(`确定要导入 ${importedData.length} 条记录吗？这将覆盖当前所有数据。`)) {
                    const previousApplications = applications;
                    applications = importedData;
                    try {
                        await saveData();
                        render();
                        closeModal();
                        try { await syncAllJobNotes(); }
                        catch (error) { alert(`数据已导入，资料路径保存失败：${error.message}。重开页面可重试。`); }
                        alert('数据导入成功！');
                    } catch (error) {
                        applications = previousApplications;
                        console.error('投递记录导入失败：', error);
                        alert(`导入失败：${error.message}`);
                    }
                }
            } else {
                alert('文件格式错误：数据不是有效的数组格式');
            }
        } catch (error) {
            alert('文件解析失败：' + error.message);
        }
    };
    reader.readAsText(file);
};
